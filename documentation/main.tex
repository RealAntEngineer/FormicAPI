\setcounter{tocdepth}{3}

% Code identifiers: monospaced, safe with underscores, and allowed to wrap at an underscore, after an opening
% parenthesis, or at a lower-to-upper case change (GpuPaddedCSRMatrixD -> Gpu|Padded|CSR|Matrix|D) instead of overflowing
% the line or being hyphenated. Needs nothing beyond the LaTeX kernel (expl3).
\ExplSyntaxOn
\tl_new:N \l__code_tl
\NewDocumentCommand{\code}{m}
{
    \tl_set:Nn \l__code_tl {#1}
    \regex_replace_all:nnN { _ } { \c{_}\c{allowbreak} } \l__code_tl
    \regex_replace_all:nnN { ([a-z0-9])([A-Z]) } { \1\c{allowbreak}\2 } \l__code_tl
    \regex_replace_all:nnN { \( } { ( \c{allowbreak} } \l__code_tl
    \texttt{\l__code_tl}
}
\ExplSyntaxOff
